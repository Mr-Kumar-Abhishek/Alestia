Crate alestia_ephem 

Source
Expand description

Orbital mechanics, analytical ephemerides (VSOP87, TASS17), sampled trajectories, JPL binary ephemerides, and Earth/planetary rotations for Alestia.

§Examples

use alestia_astro::{J2000, KeplerElements};
use alestia_ephem::{EllipticalOrbit, Orbit};

let elements = KeplerElements {
    semimajor_axis: 149_597_870.7,
    eccentricity: 0.0167,
    inclination: 0.0,
    long_ascending_node: 0.0,
    arg_pericenter: 0.0,
    mean_anomaly: 0.0,
    period: 365.25,
};
let orbit = EllipticalOrbit::new(&elements, J2000)?;
assert!(orbit.position_at_time(J2000).length() > 1.4e8);
