1//! `VSOP87` analytical planetary ephemerides.
2//!
3//! The series are Celestia's truncated `VSOP87B` (heliocentric spherical variables, planets)
4//! and `VSOP87E` (barycentric rectangular variables, Sun) tables. They are compiled in from
5//! `data/vsop87.bin`, which is generated from the upstream tables by `data/gen_vsop87.py`.
6//!
7//! Reference: Bretagnon P., Francou G., Astron. Astrophys. 202, 309 (1988).
8

21const DATA: &[u8] = include_bytes!("../data/vsop87.bin");
22const BODY_COUNT: usize = 9;
23const DAYS_PER_YEAR: f64 = 365.25;
24const DAYS_PER_MILLENNIUM: f64 = 365_250.0;
25const MAX_PREALLOC_TERMS: usize = 4096;

27/// A body with a built-in `VSOP87` series.
28#[derive(Clone, Copy, Debug, PartialEq, Eq)]
29pub enum Vsop87Body {
30    /// Mercury (heliocentric).
31    Mercury,
32    /// Venus (heliocentric).
33    Venus,
34    /// Earth (heliocentric).
35    Earth,
36    /// Mars (heliocentric).
37    Mars,
38    /// Jupiter (heliocentric).
39    Jupiter,
40    /// Saturn (heliocentric).
41    Saturn,
42    /// Uranus (heliocentric).
43    Uranus,
44    /// Neptune (heliocentric).
45    Neptune,
46    /// Sun (barycentric).
47    Sun,
48}
49
50impl Vsop87Body {
51    /// All supported bodies in table order.
52    pub const ALL: [Self; BODY_COUNT] = [
53        Self::Mercury,
54        Self::Venus,
55        Self::Earth,
56        Self::Mars,
57        Self::Jupiter,
58        Self::Saturn,
59        Self::Uranus,
60        Self::Neptune,
61        Self::Sun,
62    ];
63
64    const fn index(self) -> usize {
65        self as usize
66    }
67
68    /// Orbital period in days (`0.0` for the non-periodic Sun orbit).
69    #[must_use]
70    pub const fn period(self) -> f64 {
71        match self {
72            Self::Mercury => 0.2408 * DAYS_PER_YEAR,
73            Self::Venus => 0.6152 * DAYS_PER_YEAR,
74            Self::Earth => DAYS_PER_YEAR,
75            Self::Mars => 1.8809 * DAYS_PER_YEAR,
76            Self::Jupiter => 11.86 * DAYS_PER_YEAR,
77            Self::Saturn => 29.4577 * DAYS_PER_YEAR,
78            Self::Uranus => 84.0139 * DAYS_PER_YEAR,
79            Self::Neptune => 164.793 * DAYS_PER_YEAR,
80            Self::Sun => 0.0,
81        }
82    }
83
84    /// Conservative bound on the orbit radius in kilometers.
85    #[must_use]
86    pub const fn bounding_radius(self) -> f64 {
87        match self {
88            Self::Mercury => 60_000_000.0,
89            Self::Venus => 100_000_000.0,
90            Self::Earth => 160_000_000.0,
91            Self::Mars => 240_000_000.0,
92            Self::Jupiter => 800_000_000.0,
93            Self::Saturn => 1.5e9,
94            Self::Uranus => 3.0e9,
95            Self::Neptune => 4.7e9,
96            Self::Sun => 2_000_000.0,
97        }
98    }
99
100    /// Calendar years `(begin, end)` over which the series is used directly; outside this
101    /// window the orbit is extrapolated with a Kepler orbit.
102    #[must_use]
103    pub const fn valid_years(self) -> (i32, i32) {
104        match self {
105            Self::Sun => (-4000, 6000),
106            _ => (-4000, 4000),
107        }
108    }
109}

111/// Errors from building a `VSOP87` orbit.
112#[derive(Clone, Copy, Debug, PartialEq)]
113pub enum Vsop87Error {
114    /// The embedded series data failed to decode.
115    CorruptData,
116    /// The wrapping [`MixedOrbit`] could not be constructed.
117    Orbit(OrbitError),
118}
119
120impl fmt::Display for Vsop87Error {
121    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
122        match self {
123            Self::CorruptData => write!(f, "embedded VSOP87 series data is corrupt"),
124            Self::Orbit(err) => write!(f, "VSOP87 orbit construction failed: {err}"),
125        }
126    }
127}
128
129impl std::error::Error for Vsop87Error {
130    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
131        match self {
132            Self::Orbit(err) => Some(err),
133            Self::CorruptData => None,
134        }
135    }
136}
137
138impl From<OrbitError> for Vsop87Error {
139    fn from(err: OrbitError) -> Self {
140        Self::Orbit(err)
141    }
142}

144/// One power-of-`t` series: terms of `(A, B, C)` summed as `A * cos(B + C * t)`.
145type Series = Vec<[f64; 3]>;
146
147#[derive(Debug)]
148struct BodyTables {
149    vars: [Vec<Series>; 3],
150}
151
152fn read_count(reader: &mut BinaryReader<'_>) -> Result<usize, BinaryError> {
153    Ok(usize::try_from(reader.read_le_u32()?).unwrap_or(0))
154}
155
156fn parse_tables(bytes: &[u8]) -> Result<Vec<BodyTables>, BinaryError> {
157    let mut reader = BinaryReader::new(bytes);
158    let mut bodies = Vec::with_capacity(BODY_COUNT);
159    for _ in 0..BODY_COUNT {
160        let mut vars: [Vec<Series>; 3] = [Vec::new(), Vec::new(), Vec::new()];
161        for var in &mut vars {
162            let series_count = read_count(&mut reader)?;
163            for _ in 0..series_count {
164                let term_count = read_count(&mut reader)?;
165                let mut terms = Vec::with_capacity(term_count.min(MAX_PREALLOC_TERMS));
166                for _ in 0..term_count {
167                    let amplitude = reader.read_le_f64()?;
168                    let phase = reader.read_le_f64()?;
169                    let frequency = reader.read_le_f64()?;
170                    terms.push([amplitude, phase, frequency]);
171                }
172                var.push(terms);
173            }
174        }
175        bodies.push(BodyTables { vars });
176    }
177    Ok(bodies)
178}
179
180fn tables() -> Result<&'static [BodyTables], Vsop87Error> {
181    static TABLES: OnceLock<Option<Vec<BodyTables>>> = OnceLock::new();
182    TABLES
183        .get_or_init(|| parse_tables(DATA).ok())
184        .as_deref()
185        .ok_or(Vsop87Error::CorruptData)
186}
187
188fn sum_series(series: &[[f64; 3]], t: f64) -> f64 {
189    series
190        .iter()
191        .map(|&[amplitude, phase, frequency]| amplitude * frequency.mul_add(t, phase).cos())
192        .sum()
193}
194
195/// Evaluate `sum_k t^k * series[k](t)`.
196fn eval_variable(series: &[Series], t: f64) -> f64 {
197    let mut power = 1.0;
198    let mut total = 0.0;
199    for entry in series {
200        total += sum_series(entry, t) * power;
201        power *= t;
202    }
203    total
204}

206/// `VSOP87` orbit evaluating spherical (planets) or rectangular (Sun) series.
207#[derive(Debug)]
208pub struct Vsop87Orbit {
209    tables: &'static BodyTables,
210    rectangular: bool,
211    period: f64,
212    bounding_radius: f64,
213}
214
215impl Vsop87Orbit {
216    /// Build the raw series orbit for `body`.
217    ///
218    /// # Errors
219    /// Returns [`Vsop87Error::CorruptData`] if the embedded tables fail to decode.
220    pub fn new(body: Vsop87Body) -> Result<Self, Vsop87Error> {
221        let tables = tables()?
222            .get(body.index())
223            .ok_or(Vsop87Error::CorruptData)?;
224        Ok(Self {
225            tables,
226            rectangular: body == Vsop87Body::Sun,
227            period: body.period(),
228            bounding_radius: body.bounding_radius(),
229        })
230    }
231}
232
233impl Orbit for Vsop87Orbit {
234    fn position_at_time(&self, tdb: f64) -> DVec3 {
235        // Julian millennia since J2000.0.
236        let t = (tdb - J2000) / DAYS_PER_MILLENNIUM;
237        let [first, second, third] = &self.tables.vars;
238        if self.rectangular {
239            let x = eval_variable(first, t) * KM_PER_AU;
240            let y = eval_variable(second, t) * KM_PER_AU;
241            let z = eval_variable(third, t) * KM_PER_AU;
242            // Convert VSOP87 ecliptic axes to Celestia's internal frame.
243            DVec3::new(x, z, -y)
244        } else {
245            let longitude = eval_variable(first, t) + PI;
246            let latitude = eval_variable(second, t) - FRAC_PI_2;
247            let radius = eval_variable(third, t) * KM_PER_AU;
248            DVec3::new(
249                longitude.cos() * latitude.sin() * radius,
250                latitude.cos() * radius,
251                -longitude.sin() * latitude.sin() * radius,
252            )
253        }
254    }
255
256    fn period(&self) -> f64 {
257        self.period
258    }
259
260    fn bounding_radius(&self) -> f64 {
261        self.bounding_radius
262    }
263
264    fn is_periodic(&self) -> bool {
265        self.period > 0.0
266    }
267
268    fn sample(&self, start_time: f64, end_time: f64) -> Vec<OrbitSample> {
269        let params = if self.rectangular {
270            AdaptiveSamplingParams::for_span(end_time - start_time)
271        } else {
272            // Equal start/min/max steps make adaptive sampling behave like uniform sampling,
273            // which is much faster than the default for these smooth series.
274            let step = self.period / 150.0;
275            AdaptiveSamplingParams {
276                tolerance: 1.0,
277                start_step: step,
278                min_step: step,
279                max_step: step,
280            }
281        };
282        adaptive_sample_orbit(self, start_time, end_time, params)
283    }
284}

286/// Build the `VSOP87` orbit for `body`, valid over [`Vsop87Body::valid_years`] and
287/// extrapolated with a Kepler orbit outside that window.
288///
289/// # Errors
290/// Returns [`Vsop87Error`] if the embedded data is corrupt or the wrapper cannot be built.
291pub fn create_vsop87_orbit(body: Vsop87Body) -> Result<MixedOrbit, Vsop87Error> {
292    let orbit = Vsop87Orbit::new(body)?;
293    let (begin_year, end_year) = body.valid_years();
294    let begin = Date::new(begin_year, 1, 1).to_julian_date();
295    let end = Date::new(end_year, 1, 1).to_julian_date();
296    Ok(MixedOrbit::new(Arc::new(orbit), begin, end, SOLAR_MASS)?)
297}

299#[cfg(test)]
300mod tests {
301    use super::*;
302
303    #[test]
304    fn embedded_data_decodes_and_truncation_is_rejected() -> Result<(), Vsop87Error> {
305        let parsed = tables()?;
306        assert_eq!(parsed.len(), BODY_COUNT);
307        assert!(parse_tables(&[]).is_err());
308        assert!(parse_tables(&DATA[..DATA.len() - 1]).is_err());
309        Ok(())
310    }
311
312    #[test]
313    fn earth_matches_known_j2000_distance() -> Result<(), Vsop87Error> {
314        let orbit = Vsop87Orbit::new(Vsop87Body::Earth)?;
315        let pos = orbit.position_at_time(J2000);
316        let au = pos.length() / KM_PER_AU;
317        assert!((0.982..0.985).contains(&au), "distance {au} AU");
318        // The ecliptic normal is the Y axis in Celestia's frame.
319        assert!(pos.y.abs() < 1.0e-3 * KM_PER_AU);
320        Ok(())
321    }
322
323    #[test]
324    fn sun_is_near_barycenter() -> Result<(), Vsop87Error> {
325        let orbit = Vsop87Orbit::new(Vsop87Body::Sun)?;
326        assert!(!orbit.is_periodic());
327        let pos = orbit.position_at_time(J2000);
328        assert!(pos.is_finite());
329        assert!(pos.length() < 0.02 * KM_PER_AU);
330        Ok(())
331    }
332
333    #[test]
334    fn every_body_stays_inside_bounding_radius() -> Result<(), Vsop87Error> {
335        for body in Vsop87Body::ALL {
336            let orbit = create_vsop87_orbit(body)?;
337            let (begin, end) = body.valid_years();
338            assert!(begin < end);
339            for offset in [0.0, 1_000.0, 20_000.0] {
340                let pos = orbit.position_at_time(J2000 + offset);
341                assert!(pos.is_finite(), "{body:?} not finite");
342                // Upstream bounding radii are rounded approximations of each aphelion and are
343                // exceeded by up to ~16% (Mercury); keep them for parity and allow 20% slack.
344                assert!(
345                    pos.length() < 1.2 * body.bounding_radius(),
346                    "{body:?} outside bounding radius"
347                );
348            }
349        }
350        Ok(())
351    }
352
353    #[test]
354    fn spherical_orbit_samples_uniformly() -> Result<(), Vsop87Error> {
355        let orbit = Vsop87Orbit::new(Vsop87Body::Mercury)?;
356        let samples = orbit.sample(J2000, J2000 + orbit.period());
357        assert_ne!(samples.len(), 0);
358        Ok(())
359    }
