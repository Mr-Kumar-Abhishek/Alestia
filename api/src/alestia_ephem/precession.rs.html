181/// Ecliptic precession using the Vondrak (2006) long-period P03LP extension.
182///
183/// `centuries` is Julian centuries from J2000.0 (`(tdb - J2000) / 36525.0`).
184#[must_use]
185pub fn ecliptic_precession_p03_lp(centuries: f64) -> EclipticPole {
186    let t2 = centuries * centuries;
187    let t3 = t2 * centuries;
188
189    let mut pa = 5_750.804_069 + 0.194_831_1 * centuries - 0.000_167_39 * t2 - 4.8e-8 * t3;
190    let mut qa = -1_673.999_018 + 0.347_445_9 * centuries + 0.000_112_43 * t2 - 6.4e-8 * t3;
191
192    for term in &ECLIPTIC_PRECESSION_TERMS {
193        let theta = TAU * centuries / term.period_centuries;
194        let (sin_t, cos_t) = theta.sin_cos();
195        pa += term.p_cos * cos_t + term.p_sin * sin_t;
196        qa += term.q_cos * cos_t + term.q_sin * sin_t;
197    }
198
199    EclipticPole { pa, qa }
200}

202/// General precession and obliquity using Vondrak (2006) long-period P03LP extension.
203///
204/// `centuries` is Julian centuries from J2000.0.
205#[must_use]
206pub fn prec_obliquity_p03_lp(centuries: f64) -> PrecessionAngles {
207    let t2 = centuries * centuries;
208    let t3 = t2 * centuries;
209
210    let mut general_prec_pa =
211        7_907.295_95 + 5_044.374_034 * centuries - 0.007_134_73 * t2 + 6e-9 * t3;
212    let mut obliquity_eps_a = 83_973.876_448 - 0.042_589_9 * centuries - 0.000_001_13 * t2;
213
214    for term in &PRECESSION_TERMS {
215        let theta = TAU * centuries / term.period_centuries;
216        let (sin_t, cos_t) = theta.sin_cos();
217        general_prec_pa += term.prec_cos * cos_t + term.prec_sin * sin_t;
218        obliquity_eps_a += term.obl_cos * cos_t + term.obl_sin * sin_t;
219    }
220
221    PrecessionAngles {
222        general_prec_pa,
223        obliquity_eps_a,
224    }
225}

304#[cfg(test)]
305mod tests {
306    use super::{
307        EPS0_ARCSEC, ecliptic_precession_angles_p03, ecliptic_precession_p03,
308        ecliptic_precession_p03_lp, equatorial_precession_angles_p03, prec_obliquity_p03,
309        prec_obliquity_p03_lp,
310    };
311
312    #[test]
313    fn p03_and_p03lp_match_j2000_reference_values() -> Result<(), String> {
314        let p03 = prec_obliquity_p03(0.0);
315        if (p03.general_prec_pa).abs() > 1e-12 || (p03.obliquity_eps_a - EPS0_ARCSEC).abs() > 1e-12
316        {
317            return Err(format!("Unexpected J2000 P03 obliquity: {p03:?}"));
318        }
319
320        let p03_lp = prec_obliquity_p03_lp(0.0);
321        if (p03_lp.general_prec_pa).abs() > 0.1
322            || (p03_lp.obliquity_eps_a - EPS0_ARCSEC).abs() > 0.1
323        {
324            return Err(format!("P03LP diverged from P03 at J2000: {p03_lp:?}"));
325        }
326
327        let pole = ecliptic_precession_p03(0.0);
328        assert_eq!(pole.pa, 0.0);
329        assert_eq!(pole.qa, 0.0);
330
331        let pole_lp = ecliptic_precession_p03_lp(1.0);
332        assert!(pole_lp.pa.is_finite() && pole_lp.qa.is_finite());
333
334        let ecl = ecliptic_precession_angles_p03(1.0);
335        let eq = equatorial_precession_angles_p03(1.0);
336        if (ecl.inclination_pi_a - 46.965_35).abs() > 1e-3 || (eq.theta_a - 2003.72).abs() > 0.1 {
337            return Err(format!("Unexpected P03 1-century angles: {ecl:?}, {eq:?}"));
338        }
339        Ok(())
340    }
