1//! Lexical scanner for Celestia catalog and configuration files.
2//!
3//! Matches `celestia::util::Tokenizer` (`src/celutil/tokenizer.cpp` and `test/unit/tokenizer_test.cpp`).
4
5/// Token categories produced by [`Tokenizer`].
6#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
7pub enum TokenType {
8    /// Unquoted identifier (`[A-Za-z_][A-Za-z0-9_]*`).
9    Name,
10    /// Double-quoted string literal with escape sequences decoded.
11    String,
12    /// Integer or floating-point numeric literal.
13    Number,
14    /// Opening brace `{`.
15    BeginGroup,
16    /// Closing brace `}`.
17    EndGroup,
18    /// Opening bracket `[`.
19    BeginArray,
20    /// Closing bracket `]`.
21    EndArray,
22    /// Opening angle bracket `<` (unit block start).
23    BeginUnits,
24    /// Closing angle bracket `>` (unit block end).
25    EndUnits,
26    /// Vertical bar `|`.
27    Bar,
28    /// Equals sign `=`.
29    Equals,
30    /// Lexical error (malformed escape, unterminated string, invalid numeric prefix).
31    Error,
32    /// End of input stream.
33    End,
34}
35
36/// Byte-level lexical scanner for `.ssc`, `.stc`, `.dsc`, `.cel`, and `celestia.cfg` inputs.
37#[derive(Clone, Debug)]
38pub struct Tokenizer<'a> {
39    input: &'a [u8],
40    pos: usize,
41    line: usize,
42    pushed_back: Option<TokenType>,
43    last_token: Option<TokenType>,
44    name_buf: String,
45    string_buf: String,
46    number_f64: f64,
47    number_i64: Option<i64>,
48}
49
50impl<'a> Tokenizer<'a> {
51    /// Create a new tokenizer over a byte slice.
52    #[must_use]
53    pub const fn new(input: &'a [u8]) -> Self {
54        Self {
55            input,
56            pos: 0,
57            line: 1,
58            pushed_back: None,
59            last_token: None,
60            name_buf: String::new(),
61            string_buf: String::new(),
62            number_f64: 0.0,
63            number_i64: None,
64        }
65    }
66
67    /// Create a new tokenizer from a UTF-8 string slice.
68    #[must_use]
69    pub const fn from_str(input: &'a str) -> Self {
70        Self::new(input.as_bytes())
71    }
72
73    /// Current 1-based line number in the input.
74    #[must_use]
75    pub const fn line_number(&self) -> usize {
76        self.line
77    }
78
79    /// Push back the most recently read token so the next [`Self::next_token`] yields it again.
80    pub fn push_back(&mut self) {
81        if let Some(tok) = self.last_token {
82            self.pushed_back = Some(tok);
83        }
84    }
85
86    /// Return the identifier string if the last token was [`TokenType::Name`].
87    #[must_use]
88    pub fn name_value(&self) -> Option<&str> {
89        (self.last_token == Some(TokenType::Name)).then_some(self.name_buf.as_str())
90    }
91
92    /// Return the decoded string literal if the last token was [`TokenType::String`].
93    #[must_use]
94    pub fn string_value(&self) -> Option<&str> {
95        (self.last_token == Some(TokenType::String)).then_some(self.string_buf.as_str())
96    }
97
98    /// Return the floating-point numeric value if the last token was [`TokenType::Number`].
99    #[must_use]
100    pub fn number_f64(&self) -> Option<f64> {
101        (self.last_token == Some(TokenType::Number)).then_some(self.number_f64)
102    }
103
104    /// Return the integer numeric value if the last token was an integer [`TokenType::Number`].
105    #[must_use]
106    pub fn number_i64(&self) -> Option<i64> {
107        if self.last_token == Some(TokenType::Number) {
108            self.number_i64
109        } else {
110            None
111        }
112    }
113
114    /// Advance to and return the next [`TokenType`].
115    pub fn next_token(&mut self) -> TokenType {
116        if let Some(pushed) = self.pushed_back.take() {
117            self.last_token = Some(pushed);
118            return pushed;
119        }
120
121        self.skip_whitespace_and_comments();
122
123        let Some(&ch) = self.input.get(self.pos) else {
124            self.last_token = Some(TokenType::End);
125            return TokenType::End;
126        };
127
128        let tok = match ch {
129            b'{' => {
130                self.pos += 1;
131                TokenType::BeginGroup
132            }
133            b'}' => {
134                self.pos += 1;
135                TokenType::EndGroup
136            }
137            b'[' => {
138                self.pos += 1;
139                TokenType::BeginArray
140            }
141            b']' => {
142                self.pos += 1;
143                TokenType::EndArray
144            }
145            b'<' => {
146                self.pos += 1;
147                TokenType::BeginUnits
148            }
149            b'>' => {
150                self.pos += 1;
151                TokenType::EndUnits
152            }
153            b'|' => {
154                self.pos += 1;
155                TokenType::Bar
156            }
157            b'=' => {
158                self.pos += 1;
159                TokenType::Equals
160            }
161            b'"' => self.read_string(),
162            b'+' | b'-' | b'.' | b'0'..=b'9' => self.read_number(),
163            b'A'..=b'Z' | b'a'..=b'z' | b'_' => self.read_name(),
164            _ => {
165                self.pos += 1;
166                TokenType::Error
167            }
168        };
169
170        self.last_token = Some(tok);
171        tok
172    }
173
174    fn skip_whitespace_and_comments(&mut self) {
175        while let Some(&b) = self.input.get(self.pos) {
176            if b == b'#' {
177                self.pos += 1;
178                while let Some(&c) = self.input.get(self.pos) {
179                    self.pos += 1;
180                    if c == b'\n' {
181                        self.line += 1;
182                        break;
183                    }
184                }
185            } else if b.is_ascii_whitespace() {
186                if b == b'\n' {
187                    self.line += 1;
188                }
189                self.pos += 1;
190            } else {
191                break;
192            }
193        }
194    }
195
196    fn read_name(&mut self) -> TokenType {
197        let start = self.pos;
198        while let Some(&b) = self.input.get(self.pos) {
199            if b.is_ascii_alphanumeric() || b == b'_' {
200                self.pos += 1;
201            } else {
202                break;
203            }
204        }
205        self.name_buf.clear();
206        if let Ok(s) = std::str::from_utf8(&self.input[start..self.pos]) {
207            self.name_buf.push_str(s);
208            TokenType::Name
209        } else {
210            TokenType::Error
211        }
212    }
213
214    fn read_string(&mut self) -> TokenType {
215        self.pos += 1; // skip opening '"'
216        self.string_buf.clear();
217
218        while let Some(&b) = self.input.get(self.pos) {
219            match b {
220                b'"' => {
221                    self.pos += 1;
222                    return TokenType::String;
223                }
224                b'\\' => {
225                    self.pos += 1;
226                    let Some(&esc) = self.input.get(self.pos) else {
227                        return TokenType::Error;
228                    };
229                    self.pos += 1;
230                    match esc {
231                        b'\\' => self.string_buf.push('\\'),
232                        b'n' => self.string_buf.push('\n'),
233                        b'r' => self.string_buf.push('\r'),
234                        b't' => self.string_buf.push('\t'),
235                        b'"' => self.string_buf.push('"'),
236                        b'u' => {
237                            let Some(ch) = self.read_unicode_escape() else {
238                                return TokenType::Error;
239                            };
240                            self.string_buf.push(ch);
241                        }
242                        _ => return TokenType::Error,
243                    }
244                }
245                b'\n' => {
246                    self.line += 1;
247                    self.pos += 1;
248                    self.string_buf.push('\n');
249                }
250                0x00..=0x7F => {
251                    self.pos += 1;
252                    self.string_buf.push(char::from(b));
253                }
254                _ => {
255                    let ch = self.read_utf8_char_lossy();
256                    self.string_buf.push(ch);
257                }
258            }
259        }
260
261        TokenType::Error
262    }
263
264    fn read_unicode_escape(&mut self) -> Option<char> {
265        let hex_bytes = self.input.get(self.pos..self.pos + 4)?;
266        let hex_str = std::str::from_utf8(hex_bytes).ok()?;
267        let code = u32::from_str_radix(hex_str, 16).ok()?;
268        self.pos += 4;
269        Some(char::from_u32(code).unwrap_or('\u{FFFD}'))
270    }
271
272    /// Decode one UTF-8 scalar value matching Celestia's `UTF8Decode` behavior for invalid sequences.
273    fn read_utf8_char_lossy(&mut self) -> char {
274        let b0 = self.input[self.pos];
275        self.pos += 1;
276
277        let (expected_len, min_cp, mut cp) = match b0 {
278            0xC2..=0xDF => (2, 0x80_u32, u32::from(b0 & 0x1F)),
279            0xE0..=0xEF => (3, 0x800_u32, u32::from(b0 & 0x0F)),
280            0xF0..=0xF4 => (4, 0x1_0000_u32, u32::from(b0 & 0x07)),
281            _ => return '\u{FFFD}',
282        };
283
284        for _ in 1..expected_len {
285            let Some(&cont) = self.input.get(self.pos) else {
286                return '\u{FFFD}';
287            };
288            if cont == b'"' {
289                return '\u{FFFD}';
290            }
291            self.pos += 1;
292            if (cont & 0xC0) != 0x80 {
293                return '\u{FFFD}';
294            }
295            cp = (cp << 6) | u32::from(cont & 0x3F);
296        }
297
298        if cp < min_cp {
299            return '\u{FFFD}';
300        }
301        char::from_u32(cp).unwrap_or('\u{FFFD}')
302    }
303
304    fn read_number(&mut self) -> TokenType {
305        let start = self.pos;
306
307        if matches!(self.input.get(self.pos), Some(b'+' | b'-')) {
308            self.pos += 1;
309        }
310
311        let mut has_digits = false;
312        while let Some(&b) = self.input.get(self.pos) {
313            if b.is_ascii_digit() {
314                has_digits = true;
315                self.pos += 1;
316            } else {
317                break;
318            }
319        }
320
321        let mut is_float = false;
322        if self.input.get(self.pos) == Some(&b'.') {
323            is_float = true;
324            self.pos += 1;
325            while let Some(&b) = self.input.get(self.pos) {
326                if b.is_ascii_digit() {
327                    has_digits = true;
328                    self.pos += 1;
329                } else {
330                    break;
331                }
332            }
333        }
334
335        if !has_digits {
336            return TokenType::Error;
337        }
338
339        // Only consume exponent `[eE][+-]?[0-9]+` if at least one exponent digit exists.
340        if matches!(self.input.get(self.pos), Some(b'e' | b'E')) {
341            let mut exp_pos = self.pos + 1;
342            if matches!(self.input.get(exp_pos), Some(b'+' | b'-')) {
343                exp_pos += 1;
344            }
345            let digit_start = exp_pos;
346            while let Some(&b) = self.input.get(exp_pos) {
347                if b.is_ascii_digit() {
348                    exp_pos += 1;
349                } else {
350                    break;
351                }
352            }
353            if exp_pos > digit_start {
354                is_float = true;
355                self.pos = exp_pos;
356            }
357        }
358
359        let Ok(slice) = std::str::from_utf8(&self.input[start..self.pos]) else {
360            return TokenType::Error;
361        };
362
363        let Ok(val) = slice.parse::<f64>() else {
364            return TokenType::Error;
365        };
366        self.number_f64 = val;
367        self.number_i64 = if is_float {
368            None
369        } else {
370            slice.parse::<i64>().ok()
371        };
372        TokenType::Number
373    }
374}

380    #[test]
381    fn parses_names_and_units() {
382        let mut tok = Tokenizer::from_str("Normal Number2 Number3Number snake_case _prefixed");
383        for expected in [
384            "Normal",
385            "Number2",
386            "Number3Number",
387            "snake_case",
388            "_prefixed",
389        ] {
390            assert_eq!(tok.next_token(), TokenType::Name);
391            assert_eq!(tok.name_value(), Some(expected));
392        }
393        assert_eq!(tok.next_token(), TokenType::End);
394
395        let mut unit_tok = Tokenizer::from_str("Quantity<unit>");
396        assert_eq!(unit_tok.next_token(), TokenType::Name);
397        assert_eq!(unit_tok.name_value(), Some("Quantity"));
398        assert_eq!(unit_tok.next_token(), TokenType::BeginUnits);
399        assert_eq!(unit_tok.next_token(), TokenType::Name);
400        assert_eq!(unit_tok.name_value(), Some("unit"));
401        assert_eq!(unit_tok.next_token(), TokenType::EndUnits);
402        assert_eq!(unit_tok.next_token(), TokenType::End);
403    }

449    #[test]
450    fn parses_numbers_signs_and_trailing_exponents() -> Result<(), String> {
451        let mut tok = Tokenizer::from_str(
452            "0 0.0 12345 12345.0 32.75 1.2e6 2.3e+6 7.5e-1 1.2E6 2.3E+6 7.5E-1",
453        );
454        let expected = [
455            (0.0, Some(0)),
456            (0.0, None),
457            (12345.0, Some(12345)),
458            (12345.0, None),
459            (32.75, None),
460            (1_200_000.0, None),
461            (2_300_000.0, None),
462            (0.75, None),
463            (1_200_000.0, None),
464            (2_300_000.0, None),
465            (0.75, None),
466        ];
467        for (fval, ival) in expected {
468            assert_eq!(tok.next_token(), TokenType::Number);
469            assert_eq!(tok.number_f64(), Some(fval));
470            assert_eq!(tok.number_i64(), ival);
471        }
472        assert_eq!(tok.next_token(), TokenType::End);
473
474        let mut neg = Tokenizer::from_str("-0 -0.0 -12345 -12345.0 -32.75 -1.2e6 -2.3e+6 -7.5e-1");
475        assert_eq!(neg.next_token(), TokenType::Number);
476        let n0 = neg.number_f64().ok_or("missing -0")?;
477        assert_eq!(n0, -0.0);
478        assert!(n0.is_sign_negative());
479        assert_eq!(neg.number_i64(), Some(0));
480
481        assert_eq!(neg.next_token(), TokenType::Number);
482        let n0f = neg.number_f64().ok_or("missing -0.0")?;
483        assert_eq!(n0f, -0.0);
484        assert!(n0f.is_sign_negative());
485        assert_eq!(neg.number_i64(), None);
486
487        for invalid in ["+", "-", "+e", "+E", "-e", "-E"] {
488            let mut bad = Tokenizer::from_str(invalid);
489            assert_eq!(bad.next_token(), TokenType::Error, "input={invalid}");
490        }
491
492        for trailing in ["1.25e", "1.25E", "1.25e+", "1.25e-", "1.25E+", "1.25E-"] {
493            let mut tr = Tokenizer::from_str(trailing);
494            assert_eq!(tr.next_token(), TokenType::Number);
495            assert_eq!(tr.number_f64(), Some(1.25));
496            assert_eq!(tr.number_i64(), None);
497            assert_eq!(tr.next_token(), TokenType::Name);
498            assert_eq!(tr.name_value(), Some(&trailing[4..5]));
499        }
500
501        let mut sep = Tokenizer::from_str("123{");
502        assert_eq!(sep.next_token(), TokenType::Number);
503        assert_eq!(sep.number_i64(), Some(123));
504        assert_eq!(sep.next_token(), TokenType::BeginGroup);
505        assert_eq!(sep.next_token(), TokenType::End);
506        Ok(())
507    }
