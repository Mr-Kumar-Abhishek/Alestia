10/// Double-precision 3D parametric ray `origin + t * direction`.
11#[derive(Clone, Copy, Debug, PartialEq)]
12pub struct Ray3 {
13    /// Ray origin in 3D space.
14    pub origin: DVec3,
15    /// Ray direction vector.
16    pub direction: DVec3,
17}
18
19impl Ray3 {
20    /// Construct a new ray from `origin` and `direction`.
21    #[must_use]
22    pub const fn new(origin: DVec3, direction: DVec3) -> Self {
23        Self { origin, direction }
24    }
25
26    /// Evaluate point along the ray at parameter `t`.
27    #[must_use]
28    pub fn point_at(self, t: f64) -> DVec3 {
29        self.origin + self.direction * t
30    }
31
32    /// Transform both origin and direction by a unit quaternion.
33    #[must_use]
34    pub fn transform_quat(self, rotation: DQuat) -> Self {
35        Self::new(rotation * self.origin, rotation * self.direction)
36    }
37
38    /// Transform both origin and direction by a 3x3 matrix.
39    #[must_use]
40    pub fn transform_mat3(self, matrix: DMat3) -> Self {
41        Self::new(matrix * self.origin, matrix * self.direction)
42    }
43
44    /// Transform origin (as point, `w = 1`) and direction (as vector, `w = 0`) by a 4x4 matrix.
45    #[must_use]
46    pub fn transform_mat4(self, matrix: DMat4) -> Self {
47        Self::new(
48            matrix.transform_point3(self.origin),
49            matrix.transform_vector3(self.direction),
50        )
51    }
52
53    /// Euclidean distance from `point` to the forward half-line (`t >= 0`).
54    #[must_use]
55    pub fn distance_to_point(self, point: DVec3) -> f64 {
56        let dir_sq = self.direction.length_squared();
57        if dir_sq == 0.0 || !dir_sq.is_finite() {
58            return (point - self.origin).length();
59        }
60        let param_t = (point - self.origin).dot(self.direction) / dir_sq;
61        if param_t <= 0.0 {
62            (point - self.origin).length()
63        } else {
64            (point - self.point_at(param_t)).length()
65        }
66    }
67}

300/// Result of a forward ray intersection query including center-angle cosine.
301#[derive(Clone, Copy, Debug, PartialEq)]
302pub struct RayHit {
303    /// Nearest positive ray parameter `t` at intersection (`point_at(distance)`).
304    pub distance: f64,
305    /// Cosine of angle between ray direction and vector from ray origin to primitive center.
306    pub cos_angle_to_center: f64,
307}

309fn smallest_positive_quadratic_root(quad_a: f64, quad_b: f64, quad_c: f64) -> Option<f64> {
310    if !quad_a.is_finite() || !quad_b.is_finite() || !quad_c.is_finite() || quad_a == 0.0 {
311        return None;
312    }
313    let disc = quad_b * quad_b - quad_a * quad_c;
314    if disc < 0.0 || !disc.is_finite() {
315        return None;
316    }
317    let root_disc = disc.sqrt();
318    let sol0 = (-quad_b + root_disc) / quad_a;
319    let sol1 = (-quad_b - root_disc) / quad_a;
320
321    if sol0 > 0.0 {
322        if sol0 < sol1 || sol1 <= 0.0 {
323            Some(sol0)
324        } else {
325            Some(sol1)
326        }
327    } else if sol1 > 0.0 {
328        Some(sol1)
329    } else {
330        None
331    }
332}
333
334fn center_angle_cosine(ray: &Ray3, center: DVec3) -> f64 {
335    let to_center = center - ray.origin;
336    let norm = to_center.length();
337    if norm == 0.0 || !norm.is_finite() {
338        1.0
339    } else {
340        to_center.dot(ray.direction) / norm
341    }
342}

344/// Intersect a forward ray (`t > 0`) with a sphere, matching `celmath/intersect.h`.
345#[must_use]
346pub fn ray_sphere_intersection(ray: &Ray3, sphere: &Sphere) -> Option<f64> {
347    if sphere.radius <= 0.0 || !sphere.radius.is_finite() {
348        return None;
349    }
350    let diff = ray.origin - sphere.center;
351    let inv_rad_sq = 1.0 / (sphere.radius * sphere.radius);
352    let quad_a = ray.direction.length_squared() * inv_rad_sq;
353    let quad_b = ray.direction.dot(diff) * inv_rad_sq;
354    let quad_c = diff.length_squared() * inv_rad_sq - 1.0;
355    smallest_positive_quadratic_root(quad_a, quad_b, quad_c)
356}

358/// Intersect a forward ray with a sphere and compute `cos_angle_to_center`.
359#[must_use]
360pub fn ray_sphere_intersection_with_cos(ray: &Ray3, sphere: &Sphere) -> Option<RayHit> {
361    let distance = ray_sphere_intersection(ray, sphere)?;
362    Some(RayHit {
363        distance,
364        cos_angle_to_center: center_angle_cosine(ray, sphere.center),
365    })
366}

368/// Intersect a forward ray (`t > 0`) with an axis-aligned ellipsoid, matching `celmath/intersect.h`.
369#[must_use]
370pub fn ray_ellipsoid_intersection(ray: &Ray3, ellipsoid: &Ellipsoid) -> Option<f64> {
371    if ellipsoid.axes.x <= 0.0
372        || ellipsoid.axes.y <= 0.0
373        || ellipsoid.axes.z <= 0.0
374        || !ellipsoid.axes.is_finite()
375    {
376        return None;
377    }
378    let diff = ray.origin - ellipsoid.center;
379    let inv_axes = DVec3::ONE / ellipsoid.axes;
380    let inv_axes_sq = inv_axes * inv_axes;
381    let scaled_dir = ray.direction * inv_axes_sq;
382    let scaled_diff = diff * inv_axes_sq;
383
384    let quad_a = ray.direction.dot(scaled_dir);
385    let quad_b = ray.direction.dot(scaled_diff);
386    let quad_c = diff.dot(scaled_diff) - 1.0;
387    smallest_positive_quadratic_root(quad_a, quad_b, quad_c)
388}

421/// Determine an orientation making `-Z` point from `from` to `to` and `+Y` align with `up`.
422///
423/// Matches Celestia's `LookAt` (`celmath/geomutil.h`), returning `DQuat::IDENTITY` on degenerate inputs.
424#[must_use]
425pub fn look_at(from: DVec3, to: DVec3, up: DVec3) -> DQuat {
426    let forward = to - from;
427    let forward_len = forward.length();
428    if forward_len == 0.0 || !forward_len.is_finite() {
429        return DQuat::IDENTITY;
430    }
431    let dir = forward / forward_len;
432    let side = dir.cross(up);
433    let side_len = side.length();
434    if side_len == 0.0 || !side_len.is_finite() {
435        return DQuat::IDENTITY;
436    }
437    let right = side / side_len;
438    let ortho_up = right.cross(dir);
439    let basis = DMat3::from_cols(right, ortho_up, -dir);
440    DQuat::from_mat3(&basis).conjugate()
441}

443/// Perspective projection matrix matching Celestia's `Perspective(fovy_deg, aspect, near_z, far_z)`.
444#[must_use]
445pub fn perspective(fovy_deg: f64, aspect: f64, near_z: f64, far_z: f64) -> DMat4 {
446    if aspect == 0.0 {
447        return DMat4::IDENTITY;
448    }
449    let delta_z = far_z - near_z;
450    if delta_z == 0.0 {
451        return DMat4::IDENTITY;
452    }
453    let angle = deg_to_rad(fovy_deg * 0.5);
454    let sine = angle.sin();
455    if sine == 0.0 {
456        return DMat4::IDENTITY;
457    }
458    let cotangent = angle.cos() / sine;
459
460    DMat4::from_cols(
461        DVec4::new(cotangent / aspect, 0.0, 0.0, 0.0),
462        DVec4::new(0.0, cotangent, 0.0, 0.0),
463        DVec4::new(0.0, 0.0, -(far_z + near_z) / delta_z, -1.0),
464        DVec4::new(0.0, 0.0, -2.0 * near_z * far_z / delta_z, 0.0),
465    )
466}

468/// Orthographic projection matrix matching Celestia's `Ortho`.
469#[must_use]
470pub fn ortho(left: f64, right: f64, bottom: f64, top: f64, near_z: f64, far_z: f64) -> DMat4 {
471    let width = right - left;
472    let height = top - bottom;
473    let depth = far_z - near_z;
474    if width == 0.0 || height == 0.0 || depth == 0.0 {
475        return DMat4::IDENTITY;
476    }
477
478    DMat4::from_cols(
479        DVec4::new(2.0 / width, 0.0, 0.0, 0.0),
480        DVec4::new(0.0, 2.0 / height, 0.0, 0.0),
481        DVec4::new(0.0, 0.0, -2.0 / depth, 0.0),
482        DVec4::new(
483            -(right + left) / width,
484            -(top + bottom) / height,
485            -(far_z + near_z) / depth,
486            1.0,
487        ),
488    )
489}

497/// Project a world-space point into viewport pixel coordinates using `model_view_projection`.
498#[must_use]

501    model_view_projection: &DMat4,
502    viewport: [i32; 4],

508/// Project a world-space point through Celestia's equidistant `fisheye` dome mapping.
509#[must_use]
510pub fn project_fisheye(
511    point: DVec3,
512    model_view: &DMat4,
513    projection: &DMat4,
514    viewport: [i32; 4],
515) -> Option<DVec3> {
516    let mut eye_pos = *model_view * point.extend(1.0);
517    let radial = eye_pos.x.hypot(eye_pos.y);
518    if radial != 0.0 {
519        let phi = radial.atan2(-eye_pos.z);
520        let ratio = (phi / FRAC_PI_2) / radial;
521        eye_pos.x *= ratio;
522        eye_pos.y *= ratio;
523    }
524    let clip = *projection * eye_pos;
525    map_clip_to_viewport(clip, viewport)
526}
527
528fn map_clip_to_viewport(clip: DVec4, viewport: [i32; 4]) -> Option<DVec3> {
529    if clip.w == 0.0 || !clip.w.is_finite() {
530        return None;
531    }
532    let ndc = clip.truncate() / clip.w;
533    let normalized = DVec3::splat(0.5) + ndc * 0.5;
534    Some(DVec3::new(
535        f64::from(viewport[0]) + normalized.x * f64::from(viewport[2]),
536        f64::from(viewport[1]) + normalized.y * f64::from(viewport[3]),
537        normalized.z,
538    ))
539}

541/// Compute vertical perspective field of view in radians from viewport pixel height, screen DPI,
542/// and viewer distance to screen.
543#[must_use]
544pub fn perspective_fov(height: f64, screen_dpi: i32, distance_to_screen: i32) -> f64 {
545    if screen_dpi <= 0 || distance_to_screen <= 0 {
546        return 0.0;
547    }
548    let half_height_mm = (height / (f64::from(screen_dpi) / 25.4)) * 0.5;
549    2.0 * (half_height_mm / f64::from(distance_to_screen)).atan()
550}

605    fn ray_sphere_and_ellipsoid_intersections_find_nearest_hit() -> Result<(), String> {
606        let ray = Ray3::new(DVec3::new(-5.0, 0.0, 0.0), DVec3::X);
607        let sphere = Sphere::from_radius(2.0);
608        let Some(hit) = ray_sphere_intersection_with_cos(&ray, &sphere) else {
609            return Err("Expected sphere hit".to_owned());
610        };
611        if (hit.distance - 3.0).abs() > 1e-12 || (hit.cos_angle_to_center - 1.0).abs() > 1e-12 {
612            return Err(format!("Unexpected sphere hit: {hit:?}"));
613        }
614
615        let missed_ray = Ray3::new(DVec3::new(5.0, 0.0, 0.0), DVec3::X);
616        assert_eq!(ray_sphere_intersection(&missed_ray, &sphere), None);
617
618        let ellipsoid = Ellipsoid::from_axes(DVec3::new(3.0, 2.0, 1.0));
619        let Some(ell_hit) = ray_ellipsoid_intersection_with_cos(&ray, &ellipsoid) else {
620            return Err("Expected ellipsoid hit".to_owned());
621        };

623            return Err(format!("Unexpected ellipsoid hit: {ell_hit:?}"));
624        }
625        assert_eq!(ray_ellipsoid_intersection(&missed_ray, &ellipsoid), None);
626        Ok(())

629    #[test]
630    fn rotations_look_at_and_projections_match_expectations() -> Result<(), String> {
631        let qx = x_rotation(PI * 0.5);
632        let qy = y_rotation(PI * 0.5);
633        let qz = z_rotation(PI * 0.5);
634        if ((qx * DVec3::Y) - DVec3::Z).length() > 1e-12
635            || ((qy * DVec3::Z) - DVec3::X).length() > 1e-12
636            || ((qz * DVec3::X) - DVec3::Y).length() > 1e-12
637        {
638            return Err("Axis rotation sign mismatch".to_owned());
639        }
640
641        let orient = look_at(DVec3::ZERO, DVec3::new(0.0, 0.0, -5.0), DVec3::Y);
642        if (orient * DQuat::IDENTITY.conjugate() - DQuat::IDENTITY).length() > 1e-12 {
643            return Err(format!(
644                "Expected identity look_at along -Z, got {orient:?}"
645            ));
646        }
647
648        let proj = perspective(60.0, 16.0 / 9.0, 0.1, 100.0);
649        let viewport = [0, 0, 800, 600];
650        let Some(screen) = project_perspective(DVec3::new(0.0, 0.0, -10.0), &proj, viewport) else {
651            return Err("Expected centered projection".to_owned());
652        };
653        if (screen.x - 400.0).abs() > 1e-9 || (screen.y - 300.0).abs() > 1e-9 {
654            return Err(format!("Unexpected center pixel: {screen:?}"));
655        }
656
657        let ortho_mat = ortho(-10.0, 10.0, -10.0, 10.0, 1.0, 50.0);
658        let Some(fish) = project_fisheye(
659            DVec3::new(1.0, 0.0, -1.0),
660            &DMat4::IDENTITY,
661            &ortho_mat,
662            viewport,
663        ) else {
664            return Err("Expected fisheye projection".to_owned());
665        };
666        if !fish.is_finite() {
667            return Err("Fisheye projection produced non-finite value".to_owned());
668        }
669        let _ = ortho_2d(0.0, 800.0, 0.0, 600.0);
670        if perspective_fov(600.0, 96, 500) <= 0.0 {
671            return Err("perspective_fov must be positive".to_owned());
672        }
673        Ok(())
674    }
