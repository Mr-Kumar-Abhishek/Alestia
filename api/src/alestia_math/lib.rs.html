2//! 3D computational geometry, quaternions, frustums, intersections, and root solvers for Alestia.
3//!
4//! # Examples
5//!
6//! ```
7//! use alestia_math::{DVec3, Ray3, Sphere, ray_sphere_intersection, solve_brent};
8//!
9//! let ray = Ray3::new(DVec3::new(-5.0, 0.0, 0.0), DVec3::X);
10//! let sphere = Sphere::from_radius(2.0);
11//! assert_eq!(ray_sphere_intersection(&ray, &sphere), Some(3.0));
12//!
13//! let solved = solve_brent(|x| x * x - 2.0, 0.0, 2.0, 1e-13, 100)?;
14//! assert!((solved.root - core::f64::consts::SQRT_2).abs() < 1e-12);
15//! # Ok::<(), alestia_math::SolveError>(())
16//! ```
