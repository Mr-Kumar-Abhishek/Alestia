1//! Deterministic pseudo-random number generator matching `celmath/randutils.h`.
2
3/// Deterministic `SplitMix64` pseudo-random number generator for procedural starfields and testing.
4#[derive(Clone, Copy, Debug, PartialEq, Eq)]
5pub struct Rng {
6    state: u64,
7}
8
9impl Default for Rng {
10    fn default() -> Self {
11        Self::from_seed(0x9E37_79B9_7F4A_7C15)
12    }
13}
14
15impl Rng {
16    /// Create a generator seeded from an explicit `u64` seed.
17    #[must_use]
18    pub const fn from_seed(seed: u64) -> Self {
19        Self { state: seed }
20    }
21
22    /// Advance the state and return a 64-bit pseudo-random unsigned integer.
23    pub const fn next_u64(&mut self) -> u64 {
24        self.state = self.state.wrapping_add(0x9E37_79B9_7F4A_7C15);
25        let mut mixed = self.state;
26        mixed = (mixed ^ (mixed >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
27        mixed = (mixed ^ (mixed >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
28        mixed ^ (mixed >> 31)
29    }
30
31    /// Generate a uniform `f64` in `[0.0, 1.0)` with 53-bit significand precision.
32    pub fn next_f64(&mut self) -> f64 {
33        let bits = self.next_u64();
34        let hi_bits = u32::try_from(bits >> 37).unwrap_or(0);
35        let lo_bits = u32::try_from((bits >> 11) & 0x03FF_FFFF).unwrap_or(0);
36        (f64::from(hi_bits) * 67_108_864.0 + f64::from(lo_bits)) / 9_007_199_254_740_992.0
37    }
38
39    /// Generate a uniform `f32` in `[0.0, 1.0)` with 24-bit significand precision.
40    pub fn next_f32(&mut self) -> f32 {
41        let bits = self.next_u64();
42        let hi_bits = u16::try_from(bits >> 48).unwrap_or(0);
43        let lo_bits = u8::try_from((bits >> 40) & 0xFF).unwrap_or(0);
44        (f32::from(hi_bits) * 256.0 + f32::from(lo_bits)) / 16_777_216.0
45    }
46
47    /// Generate a uniform `f64` in `[-1.0, 1.0)`.
48    pub fn next_signed_f64(&mut self) -> f64 {
49        self.next_f64() * 2.0 - 1.0
50    }
51}

53#[cfg(test)]
54mod tests {
55    use super::Rng;
56
57    #[test]
58    fn rng_is_deterministic_and_stays_in_unit_bounds() -> Result<(), String> {
59        let mut first = Rng::from_seed(42);
60        let mut second = Rng::from_seed(42);
61        for _ in 0..128 {
62            assert_eq!(first.next_u64(), second.next_u64());
63        }
64        assert_ne!(Rng::default().next_u64(), first.next_u64());
65
66        for _ in 0..256 {
67            let val_f64 = first.next_f64();
68            let val_f32 = first.next_f32();
69            let signed = first.next_signed_f64();
70            if !(0.0..1.0).contains(&val_f64) {
71                return Err(format!("next_f64 out of bounds: {val_f64}"));
72            }
73            if !(0.0_f32..1.0_f32).contains(&val_f32) {
74                return Err(format!("next_f32 out of bounds: {val_f32}"));
75            }
76            if !(-1.0..1.0).contains(&signed) {
77                return Err(format!("next_signed_f64 out of bounds: {signed}"));
78            }
79        }
80        Ok(())
81    }
