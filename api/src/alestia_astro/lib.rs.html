2//! Astronomical timekeeping (`JD`/`UTC`/`TAI`/`TT`/`TDB`), NIST leap seconds,
3//! unit scales, coordinate frames, and photometry conversions for Alestia.
4//!
5//! # Examples
6//!
7//! ```
8//! use alestia_astro::{Date, J2000, abs_mag_to_irradiance, light_years_to_kilometers_f32};
9//!
10//! let parsed = Date::parse("2000-01-01T12:00:00Z");
11//! assert!(parsed.is_some_and(|d| (d.to_julian_date() - J2000).abs() < 1e-12));
12//!
13//! let irr = abs_mag_to_irradiance(4.81, light_years_to_kilometers_f32(32.615638));
14//! assert!(irr > 0.0);
15//! ```
