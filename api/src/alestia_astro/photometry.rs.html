1//! Photometric luminosity, visual magnitude, and Vega-normalized irradiance conversions
2//! matching `celastro/astro.h` and `celastro/astro.cpp` (`FR-009`).
3
4use crate::constants::{KM_PER_PARSEC, LN_MAG, LOWEST_IRRADIATION, LY_PER_PARSEC, SOLAR_ABSMAG};
5
6/// Luminosity of a perfectly reflective disc (used as an upper irradiance bound for culling).
7///
8/// `distance_from_sun` and `obj_radius` must be in the same length unit.
9#[must_use]
10pub fn reflected_luminosity(sun_luminosity: f32, distance_from_sun: f32, obj_radius: f32) -> f32 {
11    let ratio = obj_radius / distance_from_sun;
12    sun_luminosity * 0.25 * ratio * ratio
13}
14
15/// Distance modulus `5 * log10(distance_ly / LY_PER_PARSEC) - 5`.
16#[must_use]
17#[expect(
18    clippy::cast_possible_truncation,
19    reason = "Matches Celestia distanceModulus float32 return type"
20)]
21pub fn distance_modulus(lyrs: f64) -> f32 {
22    (5.0 * (lyrs / LY_PER_PARSEC).log10() - 5.0) as f32
23}
24
25/// Convert absolute visual magnitude and distance in light-years to apparent visual magnitude.
26#[must_use]
27pub fn abs_to_app_mag(abs_mag: f32, lyrs: f64) -> f32 {
28    abs_mag + distance_modulus(lyrs)
29}
30
31/// Convert apparent visual magnitude and distance in light-years to absolute visual magnitude.
32#[must_use]
33pub fn app_to_abs_mag(app_mag: f32, lyrs: f64) -> f32 {
34    app_mag - distance_modulus(lyrs)
35}
36
37/// Convert solar-relative visual luminosity to absolute visual magnitude.
38#[must_use]
39pub fn lum_to_abs_mag(lum: f32) -> f32 {
40    SOLAR_ABSMAG - lum.ln() * LN_MAG
41}
42
43/// Convert solar-relative visual luminosity and distance in light-years to apparent magnitude.
44#[must_use]
45pub fn lum_to_app_mag(lum: f32, lyrs: f64) -> f32 {
46    abs_to_app_mag(lum_to_abs_mag(lum), lyrs)
47}
48
49/// Convert apparent visual magnitude into Vega-normalized linear irradiance (`mag 0 => 1.0`).
50#[must_use]
51pub fn mag_to_irradiance(mag: f32) -> f32 {
52    (-mag / LN_MAG).exp()
53}
54
55/// Convert Vega-normalized linear irradiance back into apparent visual magnitude.
56#[must_use]
57pub fn irradiance_to_mag(irradiance: f32) -> f32 {
58    -irradiance.ln() * LN_MAG
59}
60
61/// Convert solar-relative visual luminosity and distance in km into Vega-normalized irradiance.
62///
63/// Retains `f64` normalization at 10 parsecs to prevent overflow on extreme luminosities/distances.
64#[must_use]
65pub fn lum_to_irradiance(lum: f32, km: f32) -> f32 {
66    lum_to_irradiance_impl(f64::from(lum), km)
67}
68
69/// Convert absolute visual magnitude to solar-relative visual luminosity.
70#[must_use]
71pub fn abs_mag_to_lum(mag: f32) -> f32 {
72    ((SOLAR_ABSMAG - mag) / LN_MAG).exp()
73}
74
75/// Convert apparent visual magnitude and distance in light-years to solar-relative luminosity.
76#[must_use]
77pub fn app_mag_to_lum(mag: f32, lyrs: f64) -> f32 {
78    abs_mag_to_lum(app_to_abs_mag(mag, lyrs))
79}
80
81/// Convert absolute visual magnitude and distance in km directly into Vega-normalized irradiance,
82/// avoiding `f32` overflow/underflow of intermediate solar luminosity.
83#[must_use]
84pub fn abs_mag_to_irradiance(mag: f32, km: f32) -> f32 {
85    let luminosity = ((f64::from(SOLAR_ABSMAG) - f64::from(mag)) / f64::from(LN_MAG)).exp();
86    lum_to_irradiance_impl(luminosity, km)
87}
88
89/// Convert faintest visible star magnitude threshold into renderer exposure scale.
90#[must_use]
91pub fn faintest_mag_to_exposure(faintest_mag: f32) -> f32 {
92    (faintest_mag / LN_MAG).exp() * LOWEST_IRRADIATION
93}
94
95/// Convert renderer exposure scale into faintest visible star magnitude threshold.
96#[must_use]
97pub fn exposure_to_faintest_mag(exposure: f32) -> f32 {
98    (exposure / LOWEST_IRRADIATION).ln() * LN_MAG
99}
100
101#[expect(
102    clippy::cast_possible_truncation,
103    reason = "Normalizes in f64 before casting to f32 matching Celestia lumToIrradianceImpl"
104)]
105fn lum_to_irradiance_impl(luminosity: f64, km: f32) -> f32 {
106    let solar_irradiance_at_10pc = f64::from(mag_to_irradiance(SOLAR_ABSMAG));
107    let distance_ratio = (10.0 * KM_PER_PARSEC) / f64::from(km);
108    (luminosity * solar_irradiance_at_10pc * distance_ratio * distance_ratio) as f32
109}

121    #[test]
122    fn distant_irradiance_follows_inverse_square_falloff() -> Result<(), String> {
123        let distance = light_years_to_kilometers_f32(1.0);
124        let reference = lum_to_irradiance(1.0, distance);
125        for scale in [1000.0_f32, 10_000.0_f32] {
126            let actual = lum_to_irradiance(1.0, distance * scale);
127            if actual <= 0.0 {
128                return Err(format!("Expected positive irradiance for scale {scale}"));
129            }
130            let product = actual * scale * scale;
131            if ((product - reference) / reference).abs() > 1e-5 {
132                return Err(format!("Inverse square drift: {product} vs {reference}"));
133            }
134        }
135        Ok(())
136    }

138    #[test]
139    fn extreme_finite_source_and_distance_retain_representable_irradiance() -> Result<(), String> {
140        let distance = light_years_to_kilometers_f32(1.0);
141        let reference = f64::from(lum_to_irradiance(1.0, distance));
142        for (lum, dist) in [(1.0e20_f32, distance), (f32::MAX, f32::MAX)] {
143            let actual = lum_to_irradiance(lum, dist);
144            if !actual.is_finite() || actual <= 0.0 {
145                return Err(format!("Non-finite/non-positive irradiance: {actual}"));
146            }
147            let dist_ratio = f64::from(dist) / f64::from(distance);
148            let scaled = f64::from(actual) / f64::from(lum) * dist_ratio * dist_ratio;
149            if ((scaled - reference) / reference).abs() > 1e-5 {
150                return Err(format!("Extreme irradiance scaled={scaled} vs {reference}"));
151            }
152        }
153        Ok(())
154    }

156    #[test]
157    fn abs_mag_to_irradiance_matches_beyond_f32_luminosity_range() -> Result<(), String> {
158        for (mag, dist_ly) in [(-100.0_f32, 10_000.0_f32), (120.0_f32, 0.000_001_f32)] {
159            let expected = mag_to_irradiance(abs_to_app_mag(mag, f64::from(dist_ly)));
160            let actual = abs_mag_to_irradiance(mag, light_years_to_kilometers_f32(dist_ly));
161            if !expected.is_finite() || !actual.is_finite() || expected <= 0.0 {
162                return Err(format!(
163                    "Invalid irradiance values: expected={expected}, actual={actual}"
164                ));
165            }
166            if ((actual / expected) - 1.0).abs() > 1e-5 {
167                return Err(format!(
168                    "abs_mag_to_irradiance ratio mismatch: {actual}/{expected}"
169                ));
170            }
171        }
172        Ok(())
173    }

175    #[test]
176    fn magnitude_luminosity_and_exposure_roundtrips_hold() -> Result<(), String> {
177        if (abs_to_app_mag(4.81, LY_PER_10PARSEC) - 4.81).abs() > 1e-5 {
178            return Err("At 10 parsecs abs and app mag must match".to_owned());
179        }
180        if (app_to_abs_mag(0.0, LY_PER_10PARSEC)).abs() > 1e-5 {
181            return Err("app_to_abs_mag at 10pc mismatch".to_owned());
182        }
183        if (lum_to_abs_mag(abs_mag_to_lum(4.81)) - 4.81).abs() > 1e-5 {
184            return Err("lum<->absmag roundtrip failed".to_owned());
185        }
186        if (app_mag_to_lum(lum_to_app_mag(2.0, 100.0), 100.0) - 2.0).abs() > 1e-4 {
187            return Err("app_mag_to_lum roundtrip failed".to_owned());
188        }
189        if (irradiance_to_mag(mag_to_irradiance(2.5)) - 2.5).abs() > 1e-5 {
190            return Err("mag<->irradiance roundtrip failed".to_owned());
191        }
192        if (exposure_to_faintest_mag(faintest_mag_to_exposure(6.5)) - 6.5).abs() > 1e-4 {
193            return Err("exposure<->faintest_mag roundtrip failed".to_owned());
194        }
195        if (reflected_luminosity(1.0, 10.0, 2.0) - 0.01).abs() > 1e-7 {
196            return Err("reflected_luminosity formula mismatch".to_owned());
197        }
198        Ok(())
199    }
