25/// Encode a floating-point coordinate using Alestia's 64.64 fixed-point [`R128`] Base64 codec.
26#[must_use]
27#[wasm_bindgen]
28pub fn r128_encode_base64(value: f64) -> String {
29    R128::from_f64(value).encode_base64()
30}
31
32/// Decode an [`R128`] Base64 coordinate string back into `f64`.
33///
34/// # Errors
35/// Returns a descriptive error message when `encoded` is not a valid `R128` Base64 string.
36#[wasm_bindgen]
37pub fn r128_decode_base64(encoded: &str) -> Result<f64, String> {
38    R128::decode_base64(encoded)
39        .map(R128::to_f64)
40        .ok_or_else(|| format!("Invalid R128 Base64 coordinate: {encoded:?}"))
41}

43/// Expand Celestia 3-letter Greek Bayer abbreviations (e.g., `"ALF Cen"` -> `"α Cen"`), one designation per line.
44#[must_use]
45#[wasm_bindgen]
46pub fn expand_greek_bayer(input: &str) -> String {
47    input
48        .lines()
49        .map(|line| replace_greek_letter_abbr(line.trim()))
50        .collect::<Vec<_>>()
51        .join("\n")
52}

54/// Parse a `#rrggbb` or `#rrggbbaa` hexadecimal color string into normalized sRGB and RGBA summary text.
55///
56/// # Errors
57/// Returns an error if `spec` is not a valid 6- or 8-digit hex color prefixed with `#`.
58#[wasm_bindgen]
59pub fn parse_color_hex(spec: &str) -> Result<String, String> {
60    let color = Color::parse_hex(spec.trim())
61        .ok_or_else(|| format!("Invalid hex color (expected #rrggbb or #rrggbbaa): {spec:?}"))?;
62    let [red, green, blue, alpha] = color.to_rgba8();
63    let linear = color.to_linear_rgba();
64    Ok(format!(
65        "hex=#{red:02x}{green:02x}{blue:02x}{alpha:02x} rgba8=({red}, {green}, {blue}, {alpha}) srgb=({:.4}, {:.4}, {:.4}, {:.4}) linear=({:.4}, {:.4}, {:.4}, {:.4})",
66        color.red(),
67        color.green(),
68        color.blue(),
69        color.alpha(),
70        linear[0],
71        linear[1],
72        linear[2],
73        linear[3],
74    ))
75}

77/// Parse a Celestia catalog block (`{ Key Value ... }`) into a formatted property summary.
78///
79/// # Errors
80/// Returns a formatted [`alestia_util::ParseError`] message if `source` is not a valid catalog hash.
81#[wasm_bindgen]
82pub fn parse_catalog_object(source: &str) -> Result<String, String> {
83    let mut parser = Parser::from_str(source);
84    let table = parser.read_hash().map_err(|err| err.to_string())?;
85    let mut output = String::new();
86    format_hash(&table, 0, &mut output);
87    Ok(output)
88}

90/// Sort newline-delimited astronomical catalog names using Celestia's natural case-insensitive sort.
91#[must_use]
92#[wasm_bindgen]
93pub fn natural_sort_lines(input: &str) -> String {
94    let mut lines: Vec<&str> = input
95        .lines()
96        .map(str::trim)
97        .filter(|line| !line.is_empty())
98        .collect();
99    lines.sort_by(|lhs, rhs| strnatcasecmp(lhs, rhs));
100    lines.join("\n")
101}

103fn format_hash(table: &AssociativeArray, indent: usize, out: &mut String) {
104    let pad = " ".repeat(indent);
105    for (key, value) in table.iter() {
106        let _ = writeln!(out, "{pad}{key}: {}", format_value(value, indent));
107    }
108}
109
110fn format_value(value: &Value, indent: usize) -> String {
111    match value {
112        Value::Number(num) => format!("{num}"),
113        Value::String(text) => format!("{text:?}"),
114        Value::Boolean(flag) => format!("{flag}"),
115        Value::Array(items) => {
116            let formatted: Vec<String> = items
117                .iter()
118                .map(|item| format_value(item, indent))
119                .collect();
120            format!("[{}]", formatted.join(", "))
121        }
122        Value::Hash(nested) => {
123            let mut buf = String::from("{\n");
124            format_hash(nested, indent + 2, &mut buf);
125            let pad = " ".repeat(indent);
126            let _ = write!(buf, "{pad}}}");
127            buf
128        }
129    }
130}

148    #[test]
149    fn r128_roundtrips_and_rejects_invalid_base64() -> Result<(), String> {
150        let encoded = r128_encode_base64(42.25);
151        let decoded = r128_decode_base64(&encoded)?;
152        if (decoded - 42.25).abs() > 1e-12 {
153            return Err(format!("R128 round-trip drifted: {decoded}"));
154        }
155        if r128_decode_base64("!!!invalid!!!").is_ok() {
156            return Err("Expected invalid Base64 input to fail".to_owned());
157        }
158        Ok(())
159    }

161    #[test]
162    fn expands_greek_bayer_abbreviations() -> Result<(), String> {
163        let single = expand_greek_bayer("XI12 Bar");
164        if single != "ξ¹² Bar" {
165            return Err(format!("Unexpected single Bayer expansion: {single}"));
166        }
167        let multi = expand_greek_bayer("ALF Cen\nXI12 Bar");
168        if multi != "α Cen\nξ¹² Bar" {
169            return Err(format!("Unexpected multiline Bayer expansion: {multi}"));
170        }
171        Ok(())
172    }
